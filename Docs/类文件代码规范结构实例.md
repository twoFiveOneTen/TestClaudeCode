下面是一个swift类文件的代码结构，需要严格遵守
```
//
//  DemoClass.swift
//  TimeMemory
//
//  Created by zhangkangkang on 2025/3/4.
//

import KKCommon
import CloudKit
import Combine

class DemoClass {

    // MARK: - Public Properties

    public var className: String = "DemoClass"

    // MARK: - Internal Properties
    
    static let shared = DemoClass()
    private(set) var currentUser: KKUserModel = KKUserModel()
    static var userId: String {
        shared.currentUser.userId
    }
    var userIdChangePublisher = CurrentValueSubject<String, Never>(userId)
    
    // MARK: - Private Properties

    private var userId: String? {
        didSet {
            fetchUserInfo()
            userIdChangePublisher.send(Self.userId)
        }
    }
    private var cancellables = Set<AnyCancellable>()
    private enum Constants {
        static let failRetryInterval: TimeInterval = 2
    }
    
    // MARK: - Init

    private init() {
    }
    
    // MARK: - Public Methods

    public func getClassName() -> String {
        className
    }

    // MARK: - Internal Methods

    func initAccount() {
        CloudkitUtil.shared.container.fetchUserRecordID { (recordID, error) in
            if let error = error {
                self.handleUserIdCloudKitError(error)
            } else if let recordID = recordID {
                self.userId = recordID.recordName
            }
        }
    }
    
    // MARK: - Private Methods

    private func handleUserIdCloudKitError(_ error: Error) {
        // 根据错误类型处理不同情况
        if let ckError = error as? CKError {
            switch ckError.code {
            case .networkUnavailable:
                monitorNetworkStatus()
            default:
                // 其他错误都延时重新获取
                DispatchQueue.main.asyncAfter(deadline: .now() + Constants.failRetryInterval) { [weak self] in
                    self?.initAccount()
                }
            }
        }
    }
    
    /// 监听网络状态，如果网络可用则重新获取用户ID
    private func monitorNetworkStatus() {
        KKNetworkTool.shared.isNetworkReachablePublisher
            .sink { [weak self] isReachable in
                guard isReachable else { return }
                self?.initAccount()
            }
            .store(in: &cancellables)
    }
    
    private func fetchUserInfo() {
        self.currentUser.userId = self.userId ?? ""
        self.currentUser.userName = generateInitUserName(userId: self.userId ?? "")
    }
    
    private func generateInitUserName(userId: String) -> String {
        guard userId.count > 8 else {
            return "用户".localized
        }
        return "用户".localized + userId.suffix(8)
    }
}

// MARK: - DemoClassAProtocol

extension DemoClass: DemoClassAProtocol {
    func demoMethod() {

    }
}
```

**解释**
1. 公共、内部、私有属性及方法、初始化方法、协议的扩展实现，都要用对应的 MARK 标签来标注，且每个类型的 MARK 标签在一个文件中有且只能有一个
2. 只提供给类内使用的属性、方法必须加private，模块内使用（internal）的属性、方法需要省略 internal 标识符
